Google Apps Script 웹앱 연동 (최근 3개월 Gmail 실시간 조회)

1단계: Google Apps Script 배포 방법 (최초 1회)
  1. script.google.com 에 접속해 새 프로젝트를 만듭니다.
  2. 아래 스크립트를 복사하여 Code.gs 에 붙여넣고 저장합니다. (보안 토큰 변경 권장)
  3. 우측 상단 배포 → 새 배포 → 유형: 웹 앱 선택 후 다음 사용자 권한으로 실행: 나, 액세스 권한이 있는 사용자: 모든 사용자로 배포합니다.
  4. 발급된 웹 앱 URL(https://script.google.com/macros/s/.../exec)을 아래에 입력하고 동기화를 실행합니다.
Apps Script 코드 (Code.gs) 
const SECRET_TOKEN = 'changhwan-mail-2026'; // 원하는 보안 토큰으로 변경 가능

function doGet(e) {
  const token = (e && e.parameter && e.parameter.token) || '';
  if (SECRET_TOKEN && token !== SECRET_TOKEN) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: '인증 토큰이 일치하지 않습니다.' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  const days = Math.min(parseInt((e && e.parameter && e.parameter.days) || '90', 10) || 90, 365);
  const maxThreads = Math.min(parseInt((e && e.parameter && e.parameter.limit) || '300', 10) || 300, 500);
  const threads = GmailApp.search('newer_than:' + days + 'd', 0, maxThreads);

  const items = threads.map(function(thread) {
    const msgs = thread.getMessages();
    const msg = msgs[msgs.length - 1];
    const labels = thread.getLabels().map(function(l) { return l.getName(); });
    if (thread.isUnread()) labels.push('UNREAD');
    return {
      id: msg.getId(),
      date: Utilities.formatDate(msg.getDate(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm'),
      sender: msg.getFrom(),
      subject: msg.getSubject() || '(제목 없음)',
      snippet: (msg.getPlainBody() || '').replace(/\s+/g, ' ').trim().slice(0, 350),
      status: thread.isUnread() ? '미처리' : '확인중',
      labels: labels.join(', ') || 'INBOX'
    };
  });

  return ContentService.createTextOutput(JSON.stringify({
    ok: true,
    syncedAt: new Date().toISOString(),
    count: items.length,
    items: items
  })).setMimeType(ContentService.MimeType.JSON);
}
